\section{Repository Structure}
\label{sec:structure}

\needspace{12\baselineskip}
\begin{lstlisting}[language={}]
weisshorn-nose-cone/
|-- nosecone/              the Python package (all the code)
|-- configs/               weisshorn.yaml + worked examples
|-- data/                  CFD database (euler_results.csv) and wall-pressure files
|-- plugins/               example plugin (Section 7.1)
|-- docs/user_guide/       this guide
|-- report/, report_650/   the nose-cone design reports
|-- CFD/                   Fluent workflow and runs (REPORT-4)
|-- tests/                 verify_reference.py + reference outputs
|-- output/                created by the tool (not in git)
\end{lstlisting}

\begin{table}[H]
\centering
\caption{Modules of the \texttt{nosecone} package, in the order of the computation chain. In Python,
\texttt{help(nosecone.<module>.<function>)} prints the documentation of any function.}
\label{tab:modules}
\footnotesize
\begin{tabular}{@{}>{\raggedright\arraybackslash}p{2.6cm}>{\raggedright\arraybackslash}p{13.4cm}@{}}
\toprule
Module & Job \\
\midrule
\texttt{config} & Read, complete and check the YAML file; every parameter with unit, range and default \\
\texttt{constants} & Physical constants and model coefficients (the only fixed numbers) \\
\texttt{geometry} & Profile families, blunting, volume, area, CP, payload length \\
\texttt{euler}, \texttt{cfd} & Axisymmetric Euler solver; CFD studies and the wave-drag database \\
\texttt{aerodynamics} & Wave-drag model: interpolation in $M$ and $L/D$, transonic rise \\
\texttt{flight} & Atmosphere, drag build-up, trajectory, calibration of $k$ \\
\texttt{stability}, \texttt{heating} & Barrowman stability and nose mass; tip and shell temperatures \\
\texttt{design} & The study: sweep, sensitivities, robustness, bluntness, decision matrix \\
\texttt{export} & Profile CSV, STL and STEP files \\
\texttt{plotting}, \texttt{reporting} & All figures; summary and LaTeX tables \\
\texttt{pipeline}, \texttt{cli} & Runs everything in order; command line \\
\bottomrule
\end{tabular}
\end{table}
